# All programme work belongs to an independent Japanese government with access to the sea.
JMC_can_manage_programme = {
 tag = A03
 exists = yes
 is_subject = no
 has_capitulated = no
 any_owned_state = { is_controlled_by = ROOT is_coastal = yes }
}

}
