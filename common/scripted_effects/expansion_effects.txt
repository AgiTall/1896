A01_begin_expansion_preparations_effect = {
 if = { limit = { country_exists = A03 NOT = { has_country_flag = A01_preparation_dispatched } NOT = { has_country_flag = A01_expansion_resolved } NOT = { has_war_with = A03 } NOT = { has_country_flag = pacific_trade_accord } }
  set_country_flag = A01_preparation_dispatched
  set_country_flag = A01_island_defence_started
  country_event = { id = A01_future.1 days = 1 }
  country_event = { id = A01_japan.2 days = 14 random_days = 16 }
  A03 = { A03_prepare_southern_war_effect = yes }
 }
}

A01_resolve_expansion_effect = {
 if = {
  limit = {
   has_country_flag = A01_expansion_campaign_started
   NOT = {
    has_country_flag = A01_expansion_resolved
   }
   has_war_with = A03
  }
  if = {
   limit = {
    A03 = {
     A03_controls_borneo_province = yes
    }
   }
   A01_borneo_treaty_effect = yes
   country_event = {
    id = expansion.3
    hours = 1
   }
  }
  else = {
   A01_expansion_victory_effect = yes
  }
 }
}
