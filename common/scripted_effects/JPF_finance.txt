# Public hooks are safe to call repeatedly; unrelated systems never edit JPF variables.
JPF_initialize_effect = {
 if = { limit = { tag = A03 NOT = { has_country_flag = JPF_initialized } }
  set_country_flag = JPF_initialized
  set_variable = { JPF_debt_pressure = 62 }
  set_variable = { JPF_budget_confidence = 34 }
  set_variable = { JPF_monthly_service = 5 }
  set_variable = { JPF_savings_used = 0 }
  add_ideas = JPF_treasury_ledger
  JPF_refresh_effect = yes
  set_country_flag = JPF_opening_queued
  country_event = { id = japan_finance.1 days = 5 }
 }
}
JPF_refresh_effect = {
 clamp_variable = { var = JPF_debt_pressure min = 0 max = 100 }
 clamp_variable = { var = JPF_budget_confidence min = 0 max = 100 }
 if = { limit = { NOT = { has_country_flag = JPF_finance_closed } }
  if = { limit = { check_variable = { var = JPF_debt_pressure value = 75 compare = greater_than_or_equals } }
   if = { limit = { NOT = { has_idea = JPF_debt_distress } } add_ideas = JPF_debt_distress }
  }
  else = { remove_ideas = JPF_debt_distress }
  if = { limit = { check_variable = { var = JPF_budget_confidence value = 30 compare = less_than } }
   if = { limit = { NOT = { has_idea = JPF_budget_mistrust } } add_ideas = JPF_budget_mistrust }
  }
  else = { remove_ideas = JPF_budget_mistrust }
  if = { limit = { has_country_flag = JPF_accounts_reformed check_variable = { var = JPF_budget_confidence value = 65 compare = greater_than_or_equals } }
   if = { limit = { NOT = { has_idea = JPF_verified_accounts } } add_ideas = JPF_verified_accounts }
  }
  else = { remove_ideas = JPF_verified_accounts }
  if = { limit = { check_variable = { var = JPF_debt_pressure value = 75 compare = greater_than_or_equals } }
   set_variable = { JPF_monthly_service = 8 }
  }
  else_if = { limit = { check_variable = { var = JPF_debt_pressure value = 50 compare = greater_than_or_equals } }
   set_variable = { JPF_monthly_service = 5 }
  }
  else = { set_variable = { JPF_monthly_service = 3 } }
 }
}

JPF_monthly_tick_effect = {
 JPF_initialize_effect = yes
 if = { limit = { NOT = { has_country_flag = JPF_finance_closed } }
  # This fixed service charge is paid in political capital, not invented currency.
  JPF_refresh_effect = yes
  if = { limit = { check_variable = { var = JPF_debt_pressure value = 75 compare = greater_than_or_equals } }
   add_political_power = -8
  }
  else_if = { limit = { check_variable = { var = JPF_debt_pressure value = 50 compare = greater_than_or_equals } }
   add_political_power = -5
  }
  else = { add_political_power = -3 }
  if = { limit = { has_war_with = A01 }
   add_to_variable = { JPF_debt_pressure = 3 }
   add_to_variable = { JPF_budget_confidence = -2 }
  }
  else_if = { limit = { has_war = yes }
   add_to_variable = { JPF_debt_pressure = 2 }
   add_to_variable = { JPF_budget_confidence = -1 }
  }
  else = {
   add_to_variable = { JPF_debt_pressure = -2 }
   add_to_variable = { JPF_budget_confidence = 1 }
  }
  if = { limit = { has_country_flag = JPC_guarantees_signed NOT = { has_country_flag = JPF_guarantee_settled } }
   add_to_variable = { JPF_debt_pressure = 1 }
  }
  if = { limit = { has_country_flag = JPF_accounts_reformed }
   add_to_variable = { JPF_debt_pressure = -1 }
   add_to_variable = { JPF_budget_confidence = 1 }
  }
  if = { limit = { has_country_flag = JPF_delivery_schedule } add_to_variable = { JPF_debt_pressure = -1 } }
  if = { limit = { has_country_flag = JPF_royalty_office has_country_flag = JPF_guarantee_settled }
   add_to_variable = { JPF_debt_pressure = -1 }
  }
  JPF_refresh_effect = yes
  if = { limit = { OR = {
   check_variable = { var = JPF_debt_pressure value = 95 compare = greater_than_or_equals }
   check_variable = { var = JPF_budget_confidence value = 10 compare = less_than_or_equals }
  } } JPF_cash_crisis_effect = yes }
  else_if = { limit = {
   check_variable = { var = JPF_debt_pressure value = 80 compare = greater_than_or_equals }
   check_variable = { var = JPF_budget_confidence value = 40 compare = less_than }
   NOT = { has_country_flag = JPF_crisis_cooldown }
  } random = { chance = 25 JPF_cash_crisis_effect = yes } }
 }
}
