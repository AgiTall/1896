# -------------------------------------------------------------------------
# sabah_war.1  --  Японский флот обнаружен у берегов Сабаха
#  Fires for A01 (Netherlands) shortly after war begins if A03 is at war.
#  Adds tension, war-support swing and optional defensive decisions.
# -------------------------------------------------------------------------
country_event = {
 id = sabah_war.1
 title = sabah_war.1.t
 desc = sabah_war.1.d
 picture = GFX_1896_event_navy
 is_triggered_only = yes
 trigger = { tag = A01 has_war_with = A03 NOT = { has_country_flag = A01_sabah_fleet_spotted } }

 immediate = { set_country_flag = A01_sabah_fleet_spotted }

 option = {
  name = sabah_war.1.a
  add_war_support = 0.05
  add_stability = -0.03
  # Defenders dig in at Sabah's main ports
  1024 = { add_building_construction = { type = coastal_bunker level = 1 instant_build = yes province = 10212 } }
 }
 option = {
  name = sabah_war.1.b
  add_war_support = 0.08
  add_stability = -0.05
  add_manpower = 500
 }
}
